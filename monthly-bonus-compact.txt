function refreshGymAssetEditor(){
const s=document.getElementById("gymAssetSelect");
if(!s)return;
const current=s.value;
s.innerHTML='<option value="">Select a challenge</option>';
[...gymsData]
.sort((a,b)=>(a.region||"").localeCompare(b.region||"")||(a.sort_order||0)-(b.sort_order||0))
.forEach(g=>s.add(new Option(`${g.region} • ${g.leader_name} • ${g.challenge_type||"Challenge"}`,g.id)));
if([...s.options].some(o=>o.value===current))s.value=current;
if(current)loadGymAssetEditor(current);
}
function loadGymAssetEditor(id){
const g=gymsData.find(x=>x.id===id);
gymLeaderImage.value=g?.leader_image_url||"";
gymBadgeImage.value=g?.badge_image_url||"";
gymLeaderImageUpload.value="";
gymBadgeImageUpload.value="";
renderGymAssetPreviews();
}
function renderGymAssetPreviews(){
const leader=gymLeaderImage?.value?.trim()||"";
const badge=gymBadgeImage?.value?.trim()||"";
if(gymLeaderPreview)gymLeaderPreview.innerHTML=leader?`<img src="${esc(leader)}" alt="Leader preview" onerror="this.parentElement.textContent='Could not load leader image'">`:"Leader preview";
if(gymBadgePreview)gymBadgePreview.innerHTML=badge?`<img src="${esc(badge)}" alt="Badge preview" onerror="this.parentElement.textContent='Could not load badge image'">`:"Badge / reward preview";
}
async function saveGymAssets(){
if(!isAdmin)return toast("Admin access required.");
const id=gymAssetSelect.value;
if(!id)return toast("Choose a gym challenge first.");
let leaderUrl=gymLeaderImage.value.trim();
let badgeUrl=gymBadgeImage.value.trim();
const leaderFile=gymLeaderImageUpload.files[0];
const badgeFile=gymBadgeImageUpload.files[0];
try{
if(leaderFile)leaderUrl=await upload(leaderFile,"gym-leader");
if(badgeFile)badgeUrl=await upload(badgeFile,"gym-badge");
}catch(e){
console.error(e);
toast("Image upload failed.");
return;
}
const{error}=await sb.from("gyms").update({
leader_image_url:leaderUrl||null,
badge_image_url:badgeUrl||null,
updated_at:new Date().toISOString()
}).eq("id",id);
if(error){
console.error(error);
toast(error.message||"Could not save gym images.");
return;
}
await loadGyms();
gymAssetSelect.value=id;
loadGymAssetEditor(id);
renderGyms();
renderBadgeCase();
toast("Gym images saved for everyone.");
}
